# ChitraMaya requirements
# Authoritative runtime dependency list. pyproject.toml only declares
# package metadata and the ChitraMaya entry point; runtime deps live here
# so we can express the cu128 wheel index, hard pins, and version
# ceilings that PEP 621 dependency syntax can't.
#
# Tested with Python 3.12, CUDA 12.8, PyTorch 2.11 (cu128), TensorRT 10.15.
# Install: pip install -r requirements.txt && pip install -e . --no-deps

--extra-index-url https://download.pytorch.org/whl/cu128

# ── PyTorch (CUDA 12.8 build) ─────────────────────────────────────────
# Hard-pinned to the cu128 wheels. The extra-index above serves them;
# pip picks them over PyPI's CPU-only wheels because the cu128 versions
# appear first in resolution.
torch==2.11.0+cu128
torchvision==0.26.0+cu128

# ── TensorRT + Torch-TensorRT ─────────────────────────────────────────
# `tensorrt` is the umbrella package; it pulls tensorrt_cu13_bindings and
# tensorrt_cu13_libs automatically. torch_tensorrt is required to load
# the BasicVSR++ sub-engines (registers __torch__.torch.classes.tensorrt
# .Engine for torch.export.load). TRT pinned to 10.15.x — 10.16 has not
# been validated against the venv stack yet.
tensorrt>=10.15.1,<10.16.0
torch_tensorrt==2.11.0

# ── CUDA runtime bindings (validate-inference helper) ─────────────────
# Used by tools/convert_models.py to allocate device buffers and streams
# for the post-build engine smoke test. Required to avoid PyTorch+TRT
# CUDA-context mixing, which produces a process-killing Access Violation
# on Blackwell (sm_120) under TRT 10.15.
cuda-python>=12.0

# ── Video codec (NVDEC / NVENC) ───────────────────────────────────────
# 2.2.0 changed Encode()/EndEncode() to return packet-dict lists where
# 2.1 returned raw bytes (field break 2026-08-16); encoder.py normalizes
# BOTH shapes (_bitstream_bytes), so the open bound stays. AV1 under 2.2
# is untested in the field -- verify CM-091 seek fix on first AV1 run.
PyNvVideoCodec>=2.1

# ── YOLO mosaic detector ──────────────────────────────────────────────
ultralytics>=8.4

# ── ONNX Runtime (for InferenceBackend = ort_cuda / ort_trt) ──────────
# Required even if you stay on the TensorRT path — used inside
# ChitraMaya/swap/engine.py for the face-swap ORT backends.
onnxruntime-gpu>=1.26

# ── Model conversion / sanitisation ───────────────────────────────────
onnx>=1.21
protobuf>=4.23

# ── Image processing ──────────────────────────────────────────────────
opencv-python>=4.13
scikit-image>=0.26
numpy>=2.4
pillow>=11.0

# ── UI / server ───────────────────────────────────────────────────────
Flask>=3.1
pywebview>=6.2

# ── CLI progress bars (mosaic pipeline + batch mode) ──────────────────
tqdm>=4.67

# ── NVML bindings (Batch 23 stall-watchdog PCIe canary) ───────────────
# Official NVIDIA NVML wrapper (imports as `pynvml`). Used to watch the
# PCIe link generation during a run: a Gen down-train preceded "GPU has
# been lost" by ~18s on a 5060 Ti field incident, so the watchdog alarms
# on it. Optional at runtime -- if missing, the canary is silently off.
nvidia-ml-py>=12.0

# ── NVIDIA Video Effects / Maxine (CM-077 RTX Super-Res secondary) ────
# Imports as `nvvfx`. Optional at RUNTIME (the pipeline warns and runs
# without the secondary if missing) but listed here so a venv rebuild
# does not silently lose SuperRes -- field event 2026-08-16: the
# post-ROCm venv rebuild dropped it and every run logged "[Secondary]
# WARNING: No module named 'nvvfx'". Plain PyPI package; needs an RTX
# GPU + recent driver at runtime, but installs anywhere.
nvidia-vfx>=0.1

# ── vs_temporalfix (CM-078 Temporal Fix): DELIBERATELY NOT LISTED ─────
# The temporal stabilizer is VENDORED, not a dependency: the
# architecture is reimplemented in
# chitramaya/mosaic/restorer/temporalfix_arch.py (credit: pifroggi,
# https://github.com/pifroggi/vs_temporalfix, Apache-2.0) and the
# weights ship in weights/. The upstream package is a VapourSynth
# plugin, is NOT on PyPI, and is never imported -- adding it here would
# make `pip install -r requirements.txt` FAIL. Do not add it.
